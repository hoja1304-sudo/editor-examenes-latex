% #1=cx, #2=cy, #3=radio, #4=anchor etiqueta A, #5=etiqueta origen (N o vacío)
\newcommand{\grafcirctrasl}[5]{%
\pgfmathtruncatemacro{\rxcoordint}{#1+#3}%
\pgfmathsetmacro{\xmin}{min(0,#1-#3)-1.8}%
\pgfmathsetmacro{\xmax}{max(0,#1+#3)+2.2}%
\pgfmathsetmacro{\ymin}{min(0,#2-#3)-1.3}%
\pgfmathsetmacro{\ymax}{max(0,#2+#3)+1.6}%
\begin{tikzpicture}[scale=0.30, baseline=(current bounding box.center)]
  \path[use as bounding box] (\xmin,\ymin) rectangle (\xmax,\ymax+1.3);
  \draw[<->, thick] (\xmin,0) -- (\xmax,0);
  \draw[<->, thick] (0,\ymin) -- (0,\ymax);
  \node[anchor=west, inner sep=1pt] at (\xmax+0.05,0) {\small\rotuloxeste};
  \node[anchor=south, inner sep=1pt] at (0,\ymax+0.05) {\small\rotuloynorte};
  \draw[thick] (#1,#2) circle (#3);
  \draw[dashed] (#1,0) -- (#1,#2);
  \draw[dashed] (\rxcoordint,0) -- (\rxcoordint,#2);
  \draw[dashed] (0,#2) -- (\rxcoordint,#2);
  \fill (#1,#2) circle (0.42);
  \node[anchor=#4, inner sep=5pt, font=\small] at (#1,#2) {$A$};
  \def\grafOrigLabel{#5}\def\grafOrigEmpty{}%
  \ifx\grafOrigLabel\grafOrigEmpty\else
    \fill (0,0) circle (0.42);
    \node[anchor=north east, inner sep=2pt, font=\small] at (0,0) {$#5$};
  \fi
  \fill (\rxcoordint,#2) circle (0.42);
  \draw (#1,-0.45) -- (#1,0.45);
  \node[anchor=north, inner sep=1.5pt] at (#1,0) {\normalsize $#1$};
  \draw (\rxcoordint,-0.45) -- (\rxcoordint,0.45);
  \node[anchor=north, inner sep=1.5pt] at (\rxcoordint,0) {\normalsize $\rxcoordint$};
  \draw (-0.45,#2) -- (0.45,#2);
  \node[anchor=east, inner sep=1.5pt] at (0,#2) {\normalsize $#2$};
\end{tikzpicture}}
