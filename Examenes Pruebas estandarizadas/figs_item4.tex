\documentclass[tikz,multi]{standalone}
\usepackage{amsmath}

\newcommand{\rotuloxeste}{\mbox{\textit{x} (este)}}
\newcommand{\rotuloynorte}{\mbox{\textit{y} (norte)}}

% #1=cx, #2=cy, #3=anchor para etiqueta A (north west / north east)
\newcommand{\grafcirc}[3]{%
\pgfmathtruncatemacro{\topY}{#2+3}%
\pgfmathsetmacro{\xmin}{min(0,#1-3)-1.8}%
\pgfmathsetmacro{\xmax}{max(0,#1+3)+2.2}%
\pgfmathsetmacro{\ymin}{min(0,#2-3)-1.3}%
\pgfmathsetmacro{\ymax}{max(0,\topY)+1.6}%
\begin{tikzpicture}[scale=0.42, baseline=(current bounding box.center)]
  \path[use as bounding box] (\xmin,\ymin) rectangle (\xmax,\ymax+1.0);
  \draw[<->, thick] (\xmin,0) -- (\xmax,0);
  \draw[<->, thick] (0,\ymin) -- (0,\ymax);
  \node[anchor=west, inner sep=1pt] at (\xmax+0.05,0) {\scriptsize\rotuloxeste};
  \node[anchor=south, inner sep=1pt] at (0,\ymax+0.05) {\scriptsize\rotuloynorte};
  \draw[dashed, black!70] (#1,0) -- (#1,#2);
  \draw[dashed, black!70] (0,#2) -- (#1,#2);
  \draw[dashed, black!70] (#1,#2) -- (#1,\topY);
  \draw[dashed, black!70] (0,\topY) -- (#1,\topY);
  \draw (#1,-0.5) -- (#1,0.5);
  \node[fill=white, anchor=north, inner sep=1.5pt] at (#1,0) {\normalsize $#1$};
  \draw (-0.5,#2) -- (0.5,#2);
  \node[fill=white, anchor=east, inner sep=1.5pt] at (0,#2) {\normalsize $#2$};
  \draw (-0.5,\topY) -- (0.5,\topY);
  \node[fill=white, anchor=east, inner sep=1.5pt] at (0,\topY) {\normalsize $\topY$};
  \draw[thick] (#1,#2) circle (3);
  \fill (#1,#2) circle (0.18);
  \node[anchor=#3, inner sep=2pt, font=\small] at (#1,#2) {$A$};
  \fill (#1,\topY) circle (0.18);
\end{tikzpicture}}

\begin{document}
\grafcirc{3}{4}{north west}
\grafcirc{-3}{-4}{north east}
\grafcirc{-3}{4}{north east}
\grafcirc{3}{-4}{north west}
\end{document}
